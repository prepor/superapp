# args:
# env: SUPERAPP_SEND_DELAY=1 SUPERAPP_MAIL_DOWN=no-route-to-host
#
# A send that cannot leave, and what the shell does about it.
#
# `SUPERAPP_MAIL_DOWN` is mail's own knob: it takes the fake servers offline
# for the whole run, so the failure here is a real one — the submission
# server refuses, the executor records it, and the problem is derived from
# the rows that stand. Nothing is mimed.
#
# The problems panel is the shell's, and it has never heard of mail: the row
# it draws, its line and its two controls all come out of mail's problem
# source as data.

wait 900

# A reply that will not go.
click "Q3 infra"
wait 800
click "reply"
wait 900
type "This one cannot leave the building."
wait 500
click "send"
wait 700

# The window runs out, the sender claims the row, the submission server
# refuses — and the problem announces itself as a toast, once.
wait 3000
shot p1-announced

key cmd 2
wait 500
type "problems"
wait 600
key enter
wait 1000
shot p2-problem

# The row's controls are data, so the shell draws a source it has never
# heard of. The link goes: a fresh sheet on the same source, which its own
# bar closes again.
click "reopen"
wait 900
shot p3-reopened
click "discard"
wait 900

# The button runs: the send is filed again, on a fresh window, so the
# condition the problem was derived from is gone with it.
click "retry"
wait 900
shot p4-retried

# …and comes back when the same letter fails again, because a problem is
# derived from rows and never stored.
wait 3000
click "retry"
wait 700
shot p5-again
quit
