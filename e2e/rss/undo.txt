wait 700
key down
wait 500
key down
wait 500

copy "The"

key cmd+z

visible "article body"

copy "Good"
key cmd+shift+z
wait 700
dblclick "article body"
copy "The"

# The restored cursor must move from that article, not skip past it.
key cmd+z
wait 700
key cmd+left
key down
wait 700
dblclick "article body"
copy "The"

# Enter only focuses the current preview; it adds no step before undo.
key cmd+left
key enter
wait 700
key cmd+z
wait 700
dblclick "article body"
copy "Good"
shot r1-undo
quit
