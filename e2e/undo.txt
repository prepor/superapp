# The undo walk: actions become history nodes; cmd+z inverts them.
wait 700
# open m1 — an action (and it marks the mail read)
click "Q3 infra"
wait 500
shot u1-opened
# archive: the panel closes, the mail leaves the inbox
click "archive"
wait 500
shot u2-archived
# undo: the panel is back, the mail is back, still the same workspace
key cmd+z
wait 600
shot u3-archive-undone
# redo: archived again
key cmd+shift+z
wait 600
shot u4-redone
# back once more, then close the panel and undo that too
key cmd+z
wait 600
key cmd+w
wait 500
key cmd+z
wait 600
shot u5-close-undone
# move the message to workspace 3, undo teleports it (and us) back
key cmd+shift+3
wait 700
key cmd+z
wait 700
shot u6-movews-undone
quit
