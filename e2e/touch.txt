# Touch gestures, driven through the same state machine android uses.
wait 900

# One-finger vertical drag on the inbox scrolls its list — and back, so the
# top rows are hittable again.
swipe "inbox" 0 -160
wait 400
shot g1-touch-scroll
swipe "inbox" 0 160
wait 400

# Widen the strip past the viewport with fresh panels…
cmdclick "Q3 infra"
wait 500
cmdclick "panel model"
wait 500
cmdclick "dentist"
wait 700
shot g2-strip-wide

# …then a two-finger pan reaches the far columns, and pans back.
pan2 -400
wait 600
shot g3-panned
pan2 400
wait 600

# A tap (a swipe going nowhere) follows a link: message opens joined.
swipe "CI failed" 2 0
wait 700
shot g4-tap-open

# Hold the help panel's header and drag it right: it re-places on drop.
holdmove "help" 500 0
wait 800
shot g5-dragged
quit
