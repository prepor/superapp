# env:

wait 800
key cmd 2
wait 300
type "terminal"
wait 300
key enter
wait 700
# A newly opened terminal accepts its first command without an extra click.
type "echo hello from terminal"
key enter
wait 300
visible "terminal output: hello from terminal"
shot terminal-half

# Returning from the launcher restores terminal input without a mouse press.
key cmd 2
wait 300
type "notes"
wait 300
key escape
wait 300
type "echo back from launcher"
key enter
wait 300
visible "terminal output: back from launcher"

# Width changes preserve the running shell and its input.
type "echo still here"
click "full width"
wait 700
key enter
wait 300
visible "terminal output: still here"
accel "half width" -
shot terminal-full
click "half width"
wait 700
visible "terminal output: still here"
accel "full width" -

# Terminal text and control keys stay in the focused panel.
click "terminal input"
type "echo mistake"
key ctrl+c
type "echo recovered"
key enter
wait 300
visible "terminal output: recovered"
dblclick "terminal output: recovered"
copy "recovered"
click "terminal input"
paste "echo pasted text\n"
wait 300
visible "terminal output: pasted text"
type "exit"
key enter
wait 300
click "restart shell"
wait 300
visible "terminal output: superapp terminal"

# Cmd-W belongs to panel closing in both width modes, even while typing.
# Undo must restore a fresh terminal; the unfinished input must be gone.
click "terminal input"
type "echo half width must close"
key cmd+w
wait 600
key cmd+z
wait 600
click "terminal input"
key cmd+a
copy "superapp terminal\n\nDemo shell: try echo, clear, or stty size.\n\n$"
click "full width"
wait 600
click "terminal input"
type "echo full width must close"
key cmd+w
wait 600
key cmd+z
wait 600
click "terminal input"
key cmd+a
copy "superapp terminal\n\nDemo shell: try echo, clear, or stty size.\n\n$"

# An inactive terminal cannot answer another panel's clipboard request.
key cmd+2

key cmd 2

wait 500
click "new note"
wait 500
click "editor"
