# args: --grid 4x3 --window 380x780
# env:
# Pick before sending, switch an existing chat from the selector, undo and
# redo the choice, walk the menu and put it away unchosen, and return to
# GLM. Every request goes through the scripted gateway.

wait 900
key cmd 2
wait 500
type "new chat"
wait 500
key enter
wait 900
type "hello"
wait 400
click "model"
wait 400
visible "model: GLM"
click "model: Sol"
wait 400
# A choice puts the caret back in the field.
type " there"
wait 400
click "send"
wait 900
click "hello there"
wait 400
click "model"
wait 400
shot model-choices
click "model: Astra"
wait 400
# The caret off the field, so cmd+z is the workspace's undo and not the
# field's. Undo brings back Sol. Redo brings back Astra.
click "Hello. I am the assistant."
wait 400
key cmd+z
wait 400
key cmd+shift+z
wait 400
# The keys walk the open menu, and esc puts it away unchosen — with the
# caret back in the field.
click "model"
wait 400
key up
wait 200
visible "model: Sol"
key escape
wait 400
absent "model: Sol"
type "again"
wait 400
click "send"
wait 900
click "again"
wait 400
click "model"
wait 400
click "model: GLM"
wait 400
# The same choice again changes nothing, and the caret still comes back.
click "model"
wait 400
click "model: GLM"
wait 400
type "same"
wait 400
click "send"
wait 900
click "same"
wait 400
shot models
quit
