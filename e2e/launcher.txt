# The launcher: double-cmd raises it, the query runs over open panels and
# the mail world; enter goes to a hit wherever it lives or opens it fresh.
wait 600
key cmd 2
wait 400
shot launcher-empty
# a subject word finds the unopened mail
type "q3"
wait 400
shot launcher-q3
# enter: a fresh un-joined trailing column on this workspace
key enter
wait 700
shot launcher-opened-mail
# park that mail on workspace 4, come back to 1
key cmd+shift+4
wait 700
key cmd+1
wait 700
# the same query is now a "go to": one hit, badged with its workspace
key cmd 2
wait 300
type "q3"
wait 400
shot launcher-q3-badged
# enter switches to workspace 4 and focuses it
key enter
wait 700
shot launcher-switched
# esc closes a fresh launcher
key cmd 2
wait 300
key esc
wait 300
# the touch path: workspaces overlay, then its search row
pan2 0 260
wait 500
shot overlay-search-row
click "search"
wait 400
type "vera"
wait 400
shot launcher-vera
# tapping the contact row opens her card here
click "Vera Kovac"
wait 700
shot launcher-contact-opened
quit
