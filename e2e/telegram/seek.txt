# Clicking halfway seeks while paused, and waiting leaves it there.
click "seek"
wait 1200
click "0:07 / 0:14"

# Dragging beyond either end clamps to the duration and zero. Moving off
# the bar vertically must keep the drag until the mouse is released.
drag "seek" 1000 60
wait 300
click "0:14 / 0:14"
drag "seek" -100 -60
wait 300
click "0:00 / 0:14"

# Seeking while playing keeps playback running from the new position.
click "play"
wait 1200
click "seek"
wait 1200
click "pause"
wait 300
click "0:08 / 0:14"

# Another click after the drag is a fresh seek and keeps the paused state.
click "seek"
wait 1200
click "0:07 / 0:14"
