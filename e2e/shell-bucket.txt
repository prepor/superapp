# args:
# env:
#
# A device gives *itself* its backup credentials. No `--bucket` flag, no file
# pushed over a cable — the endpoint is typed into the backup form and the
# bar's own *connect* is pressed, which is the only road a phone has.
#
# The clicks are the assertions: "bucket" only resolves if the form is really
# open and its field really drew, and "connect" only if the panel's bar
# carries its verb. What happens outside the app is the `bucket` file beside
# the store, which the reopened form reads back.

wait 600

# -- the form ----------------------------------------------------------------

# The backup form is the shell's own panel, a launcher root like any other.
key cmd 2
wait 500
type "backup"
wait 500
key enter
wait 800
shot bp1-form

# The three fields. Only the two public ones are ever seeded; the secret is
# write-only, so it comes up blank on a configured device too.
click "bucket"
type "https://e2e.r2.cloudflarestorage.com/superapp"
wait 300
click "key id"
type "e2e-access-key"
wait 300
click "secret"
type "e2e-cloudflare-api-token-value"
wait 300
shot bp2-filled

# -- connecting --------------------------------------------------------------

# The bar's verb: it checks the credentials, files the token where this
# platform keeps secrets, and writes the `bucket` file beside the store.
click "connect"
wait 2500
shot bp3-connected

# -- and it comes back configured -------------------------------------------

# Closed and opened again, the form shows what the device is now configured
# with: the url off the `bucket` file, and never the secret.
key cmd+w
wait 700
key cmd 2
wait 500
type "backup"
wait 500
key enter
wait 900
click "bucket"
wait 400
shot bp4-prefilled
quit
