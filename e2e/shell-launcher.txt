# args:
# env:
#
# The launcher: a double tap on cmd raises the switcher. It runs over the
# panels that are open and every app's roots — and over nothing else, since
# asking an app's own sources is a panel of its own (`shell-search`). `enter`
# goes to a hit wherever it lives, or opens it fresh where it does not. It
# never makes a second copy of something already open.
#
# The workspaces overlay's search row is the other way in: two fingers down
# raise the roster, and its search row raises this.

wait 600

key cmd 2
wait 500
shot l1-empty

# With an empty query the open panels come first, then the roots. A word
# narrows both at once: `help` is a root nothing is showing.
type "help"
wait 600
shot l2-query

# `enter`: a fresh, un-joined trailing column on this workspace. The verb on
# its bar is the assertion that the manual is what opened.
key enter
wait 900
shot l3-opened
click "try it"
wait 700

# Park it on workspace 4 and come back to 1.
key cmd+shift+4
wait 900
key cmd+1
wait 900

# The same question is a *go to* now: the panel is open, so the hit is the
# one showing it, badged with the workspace it is on.
key cmd 2
wait 500
type "help"
wait 600
shot l4-badged

# `enter` switches to workspace 4 and focuses it rather than opening a
# second copy — which the verb resolving over there is the proof of.
key enter
wait 900
click "try it"
wait 700
shot l5-switched

# `esc` closes a launcher that was just raised, and changes nothing.
key cmd 2
wait 500
key esc
wait 600

# Every app offers its own roots, so a panel nothing is showing is one query
# away: the effect log is the shell's own.
key cmd 2
wait 500
type "effects"
wait 600
key enter
wait 900
shot l6-root
click "filter"
wait 500

# -- the way in from the workspaces overlay ----------------------------------

# Two fingers down list the workspaces; its search row is a launcher entry on
# glass, and the query runs over the same two sources.
pan2 0 260
wait 700
shot l7-overlay
click "search"
wait 600
type "about"
wait 700
shot l8-from-the-row

# `enter` opens the top hit here, un-joined, exactly as it would from a
# double-cmd, and the link on the colophon is the assertion.
key enter
wait 900
click "back to help"
wait 700
shot l9-opened
quit
